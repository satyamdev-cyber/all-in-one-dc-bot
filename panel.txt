def hash_password(password):
    return hashlib.sha256(password.encode()).hexdigest()


def load_database():
    if not DATA_FILE.exists():
        data = default_database()
        save_database(data)
        return data

    try:
        with open(DATA_FILE, "r", encoding="utf-8") as f:
            return json.load(f)
    except Exception:
        return default_database()


def save_database(data):
    tmp = DATA_FILE.with_suffix(".tmp")

    with open(tmp, "w", encoding="utf-8") as f:
        json.dump(data, f, indent=4)

    tmp.replace(DATA_FILE)

def login_required(func):
    @wraps(func)
    def wrapper(*args, **kwargs):
        if not session.get("logged_in"):
            return redirect(url_for("login"))
        return func(*args, **kwargs)

    return wrapper

def get_server(server_id):
    return db["servers"].get(server_id)


def server_path(server):
    return Path(server["path"])


def is_running(server_id):
    with process_lock:
        proc = processes.get(server_id)

        if not proc:
            return False

        return proc.poll() is None


def create_server_directory(server):
    path = server_path(server)
    path.mkdir(parents=True, exist_ok=True)


def write_log(server_id, text):
    server = get_server(server_id)

    if not server:
        return

    path = server_path(server) / "console.log"

    timestamp = time.strftime("%Y-%m-%d %H:%M:%S")

    try:
        with open(path, "a", encoding="utf-8", errors="replace") as f:
            f.write(f"[{timestamp}] {text}\n")
    except Exception:
        pass


def reader_thread(server_id, proc):
    try:
        for line in iter(proc.stdout.readline, ""):
            if not line:
                break

            write_log(server_id, line.rstrip())
    except Exception as e:
        write_log(server_id, f"Output reader error: {e}")

    try:
        proc.stdout.close()
    except Exception:
        pass


def start_server_process(server_id):
    server = get_server(server_id)

    if not server:
        return False, "Server not found."

    if is_running(server_id):
        return False, "Server is already running."

    create_server_directory(server)

    command = server.get("command", "").strip()

    if not command:
        return False, "No startup command configured."

    write_log(server_id, "========================================")
    write_log(server_id, "Starting server")
    write_log(server_id, f"Command: {command}")
    write_log(server_id, "========================================")

    try:
        proc = subprocess.Popen(
            command,
            cwd=str(server_path(server)),
            shell=True,
            stdin=subprocess.PIPE,
            stdout=subprocess.PIPE,
            stderr=subprocess.STDOUT,
            text=True,
            bufsize=1
        )

        with process_lock:
            processes[server_id] = proc

        thread = threading.Thread(
            target=reader_thread,
            args=(server_id, proc),
            daemon=True
        )

        thread.start()

        return True, "Server started."

    except Exception as e:
        write_log(server_id, f"Failed to start: {e}")
        return False, str(e)


def stop_server_process(server_id):
    server = get_server(server_id)

    if not server:
        return False, "Server not found."

    with process_lock:
        proc = processes.get(server_id)

    if not proc or proc.poll() is not None:
        return False, "Server is not running."

    # Try Minecraft's normal stop command first.
    try:
        proc.stdin.write("stop\n")
        proc.stdin.flush()

        try:
            proc.wait(timeout=15)
        except subprocess.TimeoutExpired:
            proc.terminate()

            try:
                proc.wait(timeout=5)
            except subprocess.TimeoutExpired:
                proc.kill()

    except Exception:
        try:
            proc.terminate()
        except Exception:
            pass

    with process_lock:
        processes.pop(server_id, None)

    write_log(server_id, "Server stopped.")

    return True, "Server stopped."


def restart_server_process(server_id):
    ok, msg = stop_server_process(server_id)

    if not ok and "not running" not in msg.lower():
        return False, msg

    time.sleep(2)

    return start_server_process(server_id)


def send_command(server_id, command):
    with process_lock:
        proc = processes.get(server_id)

    if not proc or proc.poll() is not None:
        return False, "Server is not running."

    try:
        proc.stdin.write(command + "\n")
        proc.stdin.flush()

        write_log(server_id, f"> {command}")

        return True, "Command sent."

    except Exception as e:
        return False, str(e)


# ============================================================
# SAFE PATH HANDLING
# ============================================================

def safe_path(server, relative):
    root = server_path(server).resolve()

    requested = (root / relative).resolve()

    if requested != root and root not in requested.parents:
        abort(403)

    return requested

@app.route("/api/server/<server_id>/console")
@login_required
def api_console(server_id):

    server = get_server(server_id)

    if not server:
        return jsonify({
            "console": "Server not found."
        }), 404

    log_file = server_path(server) / "console.log"

    if not log_file.exists():
        return jsonify({
            "console": "No console output yet."
        })

    try:

        with open(
            log_file,
            "r",
            encoding="utf-8",
            errors="replace"
        ) as f:

            content = f.read()

        # Keep browser response reasonable.
        content = content[-100000:]

        return jsonify({
            "console": content
        })

    except Exception as e:

        return jsonify({
            "console": str(e)
        })

@app.route("/server/<server_id>/files")
@app.route("/server/<server_id>/files/<path:relative>")
@login_required
def files(server_id, relative=""):

    server = get_server(server_id)

    if not server:
        return "Server not found", 404

    current = safe_path(server, relative)

    if not current.exists():
        return "Path does not exist", 404

    if not current.is_dir():
        return send_file(current)

    entries = []

    for item in sorted(
        current.iterdir(),
        key=lambda x: (not x.is_dir(), x.name.lower())
    ):

        rel = item.relative_to(
            server_path(server)
        )

        entries.append({
            "name": item.name,
            "path": str(rel),
            "directory": item.is_dir(),
            "size": item.stat().st_size if item.is_file() else ""
        })

    rows = ""

    parent = Path(relative).parent

    if relative:
        parent_url = url_for(
            "files",
            server_id=server_id,
            relative="" if str(parent) == "." else str(parent)
        )

        rows += f"""
        <tr>
            <td colspan="3">
                <a class="btn btn-gray"
                   href="{parent_url}">
                    .. Parent
                </a>
            </td>
        </tr>
        """

    for item in entries:

        if item["directory"]:

            href = url_for(
                "files",
                server_id=server_id,
                relative=item["path"]
            )

            name = "📁 " + item["name"]

        else:

            href = url_for(
                "download_file",
                server_id=server_id,
                relative=item["path"]
            )

            name = "📄 " + item["name"]

        rows += f"""
        <tr>
            <td>
                <a href="{href}">
                    {name}
                </a>
            </td>
            <td>{item["size"]}</td>
            <td>
                <button
                    onclick="deleteFile('{item["path"]}')"
                    class="btn btn-danger">
                    Delete
                </button>
            </td>
        </tr>
        """

    html = """
    <!DOCTYPE html>
    <html>
    <head>
        <title>Files</title>
    """ + STYLE + """
    </head>

    <body>

    <div class="nav">
        <div class="logo">File Manager</div>
        <a class="btn btn-gray"
           href="/server/""" + server_id + """">
           Back
        </a>
    </div>

    <div class="container">

    <div class="card">

        <h2>Upload File</h2>

        <form method="POST"
              action="/server/""" + server_id + """/upload"
              enctype="multipart/form-data">

            <input type="hidden"
                   name="path"
                   value=\"""" + relative + """\">

            <input type="file" name="file" required>

            <button type="submit">
                Upload
            </button>

        </form>

    </div>

    <div class="card">

        <h2>
            /""" + relative + """

@app.route("/server/<server_id>/download/<path:relative>")
@login_required
def download_file(server_id, relative):

    server = get_server(server_id)

    if not server:
        return "Server not found", 404

    path = safe_path(server, relative)

    if not path.exists() or not path.is_file():
        return "File not found", 404

@app.route("/server/<server_id>/upload", methods=["POST"])
@login_required
def upload_file(server_id):

    server = get_server(server_id)

    if not server:
        return "Server not found", 404

    uploaded = request.files.get("file")
    relative = request.form.get("path", "")

    if not uploaded:
        return "No file selected", 400

    directory = safe_path(server, relative)

    if not directory.exists() or not directory.is_dir():
        return "Invalid directory", 400

    filename = Path(uploaded.filename).name

    if not filename:
        return "Invalid filename", 400

    destination = safe_path(
        server,
        str(Path(relative) / filename)
    )

    uploaded.save(destination)

    return redirect(
        url_for(
            "files",
            server_id=server_id,
            relative=relative
        )
    )


@app.route("/server/<server_id>/delete", methods=["POST"])
@login_required
def delete_file(server_id):

    server = get_server(server_id)

    if not server:
        return jsonify({
            "success": False
        }), 404

    data = request.get_json(silent=True) or {}

    relative = data.get("path", "")

    if not relative:
        return jsonify({
            "success": False
        }), 400

    path = safe_path(server, relative)

    root = server_path(server).resolve()

    if path == root:
        return jsonify({
            "success": False,
            "message": "Cannot delete server root."
        }), 403

    try:

        if path.is_dir():
            shutil.rmtree(path)

        elif path.exists():
            path.unlink()

        return jsonify({
            "success": True
        })

    except Exception as e:

        return jsonify({
            "success": False,
            "message": str(e)
        }), 500

@app.route("/server/<server_id>/delete-server", methods=["POST"])
@login_required
def delete_server(server_id):

    server = get_server(server_id)

    if not server:
        return "Server not found", 404

    if is_running(server_id):
        stop_server_process(server_id)

    path = server_path(server)

    try:

        if path.exists():
            shutil.rmtree(path)

    except Exception as e:
        return f"Could not delete files: {e}", 500

    db["servers"].pop(server_id, None)

    save_database(db)

    return redirect(url_for("dashboard"))
